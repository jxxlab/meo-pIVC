# ============================================================
# Figure 3a:
#   UMAP split into three groups:

#   Vitro1 = RD1-RD4   (meo-pIVC1)

# ============================================================

# ============================================================

# ============================================================
# 1. Define the three Figure 3 groups
# ============================================================
# This grouping is directly taken from the HTML notebook.

pbmc$sample2 <- ifelse(
  pbmc$sample1 %in% c("E9", "E10", "E11", "E12"),
  "Vivo",
  ifelse(
    pbmc$sample1 %in% c("RD1", "RD2", "RD3", "RD4"),
    "Vitro1",
    ifelse(
      pbmc$sample1 %in% c("D1", "D2", "D3", "D4", "D5", "D6"),
      "Vitro2",
      NA
    )
  )
)

pbmc$sample2 <- factor(
  pbmc$sample2,
  levels = c("Vivo", "Vitro1", "Vitro2")
)

# ============================================================
# Figure 3b
# Calculate CX1 cell percentage for every stage/day
# ============================================================
# Original HTML:
# meta = as.data.frame(as.array(table(pbmc$sample1,pbmc$CX1)))
# colnames(meta) = c("Sample","Group","Cells")
#
# Then:
# Fraction = Cells / sum(Cells) within each Sample
# Fraction = Fraction * 100

meta <- as.data.frame(
  as.array(
    table(
      pbmc$sample1,
      pbmc$CX1
    )
  )
)

colnames(meta) <- c(
  "Sample",
  "Group",
  "Cells"
)

head(meta)

# Fraction of each CX1 cluster within each developmental stage/day
meta <- plyr::ddply(
  meta,
  "Sample",
  transform,
  Fraction = Cells / sum(Cells)
)

# Convert to percentage
meta$Fraction <- meta$Fraction * 100

head(meta)
head(meta$Fraction)

# Check that every stage/day sums to 100%
aggregate(
  Fraction ~ Sample,
  data = meta,
  FUN = sum
)


# ============================================================
# 3b. Define CX1 order and colors
# ============================================================
# Figure 3 contains CX1 clusters 1-44.
grp_order <- as.character(44:1)

meta$Group <- factor(
  meta$Group,
  levels = grp_order
)

# Same first 44 colors as Figure 3a
cols <- setNames(
  x[1:44],
  as.character(1:44)
)


# ============================================================
# 3b. Fix sample order
# ============================================================
# Because Figure 3b is horizontal, factor levels are specified
# from bottom to top.
#
# Bottom:
#   in vivo E9-E12
# Middle:
#   meo-pIVC1 RD1-RD4
# Top:
#   meo-pIVC2 D1-D6
#
# This follows the ordering logic present in the HTML.

meta$Sample <- factor(
  meta$Sample,
  levels = c(
    "E12", "E11", "E10", "E9",
    "RD4", "RD3", "RD2", "RD1",
    "D6", "D5", "D4", "D3", "D2", "D1"
  )
)


# ============================================================
# 3b. Plot stacked percentage bars
# ============================================================

options(
  repr.plot.width = 20,
  repr.plot.height = 10
)

p3b <- ggplot(
  meta,
  aes(
    x = Fraction,
    y = Sample,
    fill = Group
  )
) +
  geom_bar(
    stat = "identity",
    position = position_stack(
      reverse = FALSE
    ),
    color = NA,
    width = 0.7,
    linewidth = 0.25
  ) +
  theme_bw() +
  theme(
    axis.title.x = element_text(
      size = 20,
      face = "bold"
    ),
    axis.text.x = element_text(
      size = 20
    ),
    axis.text.y = element_text(
      size = 20
    ),
    legend.text = element_text(
      size = 12
    ),
    legend.title = element_blank(),
    panel.grid.major = element_blank(),
    panel.grid.minor = element_blank()
  ) +
  xlab(
    "Percentage of cells (%)"
  ) +
  ylab("") +
  scale_fill_manual(
    values = cols,
    breaks = grp_order,
    guide = guide_legend(
      reverse = TRUE,
      keywidth = 2,
      keyheight = 2
    )
  )

p3b

ggsave(
  "Figure3b_CX1_cell_percentage.pdf",
  plot = p3b,
  width = 20,
  height = 10,
  dpi = 300,
  limitsize = FALSE
)
