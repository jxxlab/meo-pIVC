#============================================================
# 2. 定义KEGG GSEA函数
#============================================================

run_gsea_kegg <- function(
    res_df,
    id_col = "Geneid",
    from_type = "ENSEMBL",
    score_col = "stat",
    organism = "mmu"
) {

  #----------------------------------------------------------
  # 1. 整理差异分析结果
  #----------------------------------------------------------

  df <- as.data.frame(
    res_df,
    stringsAsFactors = FALSE
  )

  if (!id_col %in% colnames(df)) {
    stop(
      "结果中没有找到基因ID列：",
      id_col
    )
  }

  # 若没有stat，则使用log2FoldChange
  if (!score_col %in% colnames(df)) {

    if ("log2FoldChange" %in% colnames(df)) {

      score_col <- "log2FoldChange"

      message(
        "没有找到stat，改用log2FoldChange排序。"
      )

    } else {

      stop(
        "结果中既没有stat，也没有log2FoldChange。"
      )
    }
  }

  df <- df %>%
    dplyr::filter(
      !is.na(.data[[id_col]]),
      .data[[id_col]] != "",
      !is.na(.data[[score_col]]),
      is.finite(.data[[score_col]])
    )

  # 去除ENSEMBL版本号
  df[[id_col]] <- sub(
    "\\..*$",
    "",
    as.character(df[[id_col]])
  )

  #----------------------------------------------------------
  # 2. ENSEMBL转换成ENTREZID
  #----------------------------------------------------------

  gene_map <- suppressMessages(
    clusterProfiler::bitr(
      unique(df[[id_col]]),
      fromType = from_type,
      toType = "ENTREZID",
      OrgDb = org.Mm.eg.db
    )
  )

  if (nrow(gene_map) == 0) {
    stop("没有基因成功转换为ENTREZID。")
  }

  # Geneid对应gene_map中的ENSEMBL
  join_by <- stats::setNames(
    from_type,
    id_col
  )

  df2 <- df %>%
    dplyr::inner_join(
      gene_map,
      by = join_by
    )

  #----------------------------------------------------------
  # 3. 一个ENTREZID只保留一个排序值
  #----------------------------------------------------------

  df2 <- df2 %>%
    dplyr::group_by(
      ENTREZID
    ) %>%
    dplyr::slice_max(
      order_by = abs(.data[[score_col]]),
      n = 1,
      with_ties = FALSE
    ) %>%
    dplyr::ungroup()

  #----------------------------------------------------------
  # 4. 构建GSEA排序向量
  #----------------------------------------------------------

  gene_list <- df2[[score_col]]

  names(gene_list) <- as.character(
    df2$ENTREZID
  )

  gene_list <- sort(
    gene_list,
    decreasing = TRUE
  )

  # 删除可能重复的名字
  gene_list <- gene_list[
    !duplicated(names(gene_list))
  ]

  message(
    "进入GSEA的基因数量：",
    length(gene_list)
  )

  #----------------------------------------------------------
  # 5. KEGG GSEA
  #----------------------------------------------------------

  gsea_kegg <- clusterProfiler::gseKEGG(
    geneList = gene_list,
    organism = organism,
    keyType = "ncbi-geneid",
    minGSSize = 10,
    maxGSSize = 500,
    pvalueCutoff = 1,
    pAdjustMethod = "BH",
    verbose = FALSE,
    seed = TRUE
  )

  gsea_df <- as.data.frame(
    gsea_kegg
  )

  list(
    gsea = gsea_kegg,
    table = gsea_df,
    gene_list = gene_list,
    mapped_table = df2,
    score_col = score_col
  )
}

#============================================================
# 4. 整理并合并所有GSEA结果
#============================================================

prepare_gsea_table <- function(
    gsea_result,
    treatment,
    tissue,
    comparison
) {

  result_df <- as.data.frame(
    gsea_result$table,
    stringsAsFactors = FALSE
  )

  if (nrow(result_df) == 0) {
    return(data.frame())
  }

  result_df %>%
    dplyr::mutate(
      Treatment = treatment,
      Tissue = tissue,
      Comparison = comparison,

      Description_clean = stringr::str_remove(
        Description,
        " - Mus musculus.*$"
      ),

      Direction = dplyr::case_when(
        NES > 0 ~ paste0(
          treatment,
          " enriched"
        ),

        NES < 0 ~ "Control enriched",

        TRUE ~ "No direction"
      ),

      neg_log10_FDR = -log10(
        pmax(
          p.adjust,
          .Machine$double.xmin
        )
      )
    )
}

row_z <- function(m) {
  z <- t(scale(t(m))); z[!is.finite(z)] <- 0

  z

                     TGFB=classic_gene_sets_30$TGFB,NODAL=classic_gene_sets_30$NODAL)
mat_group <- sapply(group_all,function(g) rowMeans(expr_mat[,coldata$Group==g,drop=FALSE]))
mat_group_z <- row_z(mat_group)
ann <- data.frame(Tissue=c("YS","Emb","YS","Emb"),Condition=c("Ctrl","Ctrl","RA","RA"),row.names=group_show)
ann_colors <- list(Tissue=c(YS="#00A087",Emb="#F39B7F"),
                   Condition=c(Ctrl="#4DBBD5",RA="#E64B35"))
hm_col <- colorRampPalette(c("#2C7FB8","#7FCDBB","#FFFFBF","#FDAE61","#D73027"))(100)
letters_panel <- c(RA="e",HOX="f",TGFB="g",NODAL="h")
match_report <- bind_rows(lapply(names(heatmap_sets),function(pw) {
  genes <- unique(heatmap_sets[[pw]])
  data.frame(Pathway=pw,Gene=genes,Found=genes %in% rownames(mat_group_z))
}))
write.csv(match_report,file.path(out_dir,"Figure4e_h_gene_match_report.csv"),row.names=FALSE)
for(pw in names(heatmap_sets)) {
  genes <- intersect(unique(heatmap_sets[[pw]]),rownames(mat_group_z))
  if(length(genes)<2) stop("Too few matched heatmap genes: ",pw)
  m <- mat_group_z[genes,group_show,drop=FALSE]
  title <- paste0(ifelse(pw=="TGFB","TGFβ",pw)," pathway")
  p <- pheatmap(m,color=hm_col,breaks=seq(-2.5,2.5,length.out=101),
    annotation_col=ann,annotation_colors=ann_colors,
    cluster_rows=TRUE,cluster_cols=FALSE,treeheight_row=0,treeheight_col=0,
    show_rownames=TRUE,show_colnames=FALSE,border_color=NA,
    cellwidth=18,cellheight=10,fontsize_row=8,main=title,silent=TRUE)
  prefix <- paste0("Figure4",letters_panel[[pw]],"_",pw)
  grDevices::pdf(file.path(out_dir,paste0(prefix,".pdf")),width=5.5,height=max(5,nrow(m)*10/72+2.5))
  grid::grid.newpage(); grid::grid.draw(p$gtable); grDevices::dev.off()
  write.csv(m,file.path(out_dir,paste0(prefix,"_matrix.csv")))
  write.csv(data.frame(Gene=rownames(m)[p$tree_row$order]),
            file.path(out_dir,paste0(prefix,"_row_order.csv")),row.names=FALSE)
}
