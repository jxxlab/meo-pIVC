# ============================================================
# PIVC1 glycolysis and OxPhos/TCA metabolite heatmaps
# Figure-only script
#
# R version: 4.1.3
# Packages: readxl 1.4.1; dplyr 1.2.1; stringr 1.5.0;
#           ComplexHeatmap 2.10.0
#
# Input:
#   data/all_sample_data_tm36.xlsx
#
# Output:
#   results/PIVC1_Glycolysis_heatmap.pdf
#   results/PIVC1_OxPhos_TCA_heatmap.pdf
# ============================================================

suppressPackageStartupMessages({
  library(readxl); library(dplyr); library(stringr); library(ComplexHeatmap)
})

input_file <- "data/all_sample_data_tm36.xlsx"
output_dir <- "results"
dir.create(output_dir, recursive=TRUE, showWarnings=FALSE)

glycolysis_mets <- c(
  "Glucose","D-Glucose 6-Phosphate","Fructose-6-phosphate",
  "1,6-di-O-phosphono-beta-D-fructofuranose","Dl-Glyceraldehyde 3-Phosphate",
  "3-Phosphoglyceric acid","Glycerol 3-Phosphate","Phosphoenolpyruvate","Lactic acid"
)

oxphos_tca_mets <- c(
  "Citric Acid","Aconitic Acid","α-Ketoglutaric Acid (α-KG)",
  "Succinic Acid","Fumaric Acid","Malic acid","Oxaloacetic acid"
)

col_culture <- c("In vivo"="#1b9e77","In vitro"="#d95f02")
col_stage <- c("11"="#4DBBD5","12"="#E64B35")
five_cols <- c("#2C7FB8","#7FCDBB","#FFFFBF","#FDAE61","#D73027")

dat <- read_excel(input_file) %>% as.data.frame(check.names=FALSE)
if(!"Compounds" %in% colnames(dat)) stop("Column 'Compounds' was not found.")

sample_cols <- grep("^[ED][0-9]+E[0-9]+$", colnames(dat), value=TRUE)
if(length(sample_cols) == 0) stop("No PIVC1 embryo sample columns were detected.")

si <- data.frame(Sample=sample_cols, stringsAsFactors=FALSE) %>%
  mutate(
    Culture=ifelse(str_starts(Sample,"E"),"In vivo","In vitro"),
    Stage=str_extract(Sample,"[0-9]+"),
    Replicate=as.integer(str_extract(Sample,"[0-9]+$"))
  ) %>%
  filter(Stage %in% c("11","12")) %>%
  arrange(factor(Culture, levels=c("In vivo","In vitro")),
          factor(Stage, levels=c("11","12")), Replicate)

mat <- dat %>% select(all_of(si$Sample)) %>% as.matrix()
storage.mode(mat) <- "numeric"
rownames(mat) <- make.unique(as.character(dat$Compounds))
mat_log <- log2(mat + 1)
mat_log <- mat_log[, si$Sample, drop=FALSE]

plot_pathway_heatmap <- function(metabolites, title_text, output_file, pdf_width, pdf_height) {
  met_present <- metabolites[metabolites %in% rownames(mat_log)]
  met_missing <- setdiff(metabolites, met_present)
  if(length(met_missing) > 0) message("Skipped metabolites not found: ", paste(met_missing, collapse="; "))
  if(length(met_present) == 0) stop("None of the requested metabolites were found for ", title_text, ".")

  si_sub <- si %>%
    mutate(Group=paste0(ifelse(Culture=="In vivo","E","D"), Stage)) %>%
    filter(Group %in% c("E11","E12","D11","D12"))

  group_order <- c("E11","E12","D11","D12")
  mat_sub <- mat_log[met_present, si_sub$Sample, drop=FALSE]
  mat_mean <- sapply(group_order, function(g) {
    smp <- si_sub$Sample[si_sub$Group == g]
    if(length(smp) == 0) rep(NA_real_, nrow(mat_sub)) else rowMeans(mat_sub[, smp, drop=FALSE], na.rm=TRUE)
  })
  mat_mean <- as.matrix(mat_mean)
  rownames(mat_mean) <- met_present
  colnames(mat_mean) <- group_order

  mat_z <- t(scale(t(mat_mean)))
  mat_z[!is.finite(mat_z)] <- 0
  mat_z[mat_z > 2] <- 2
  mat_z[mat_z < -2] <- -2

  col_fun <- circlize::colorRamp2(c(-2,-1,0,1,2), five_cols)

  ha <- HeatmapAnnotation(
    Culture=c("In vivo","In vivo","In vitro","In vitro"),
    Stage=c("11","12","11","12"),
    col=list(Culture=col_culture, Stage=col_stage),
    show_annotation_name=TRUE,
    annotation_name_gp=grid::gpar(fontsize=13, fontface="bold"),
    simple_anno_size=grid::unit(0.45,"cm")
  )

  cell_size <- 0.85
  ht <- Heatmap(
    mat_z, name="Z-score", col=col_fun, cluster_rows=FALSE, cluster_columns=FALSE,
    show_row_names=TRUE, show_column_names=TRUE, row_names_side="right",
    row_names_gp=grid::gpar(fontsize=13), column_names_gp=grid::gpar(fontsize=13),
    column_title=title_text, column_title_gp=grid::gpar(fontsize=16, fontface="bold"),
    width=grid::unit(ncol(mat_z)*cell_size,"cm"),
    height=grid::unit(nrow(mat_z)*cell_size,"cm"),
    top_annotation=ha,
    heatmap_legend_param=list(
      at=c(-2,-1,0,1,2), direction="horizontal", title_position="topcenter",
      labels_gp=grid::gpar(fontsize=12), title_gp=grid::gpar(fontsize=13, fontface="bold")
    )
  )

  grDevices::pdf(output_file, width=pdf_width, height=pdf_height, useDingbats=FALSE)
  draw(ht, heatmap_legend_side="left", annotation_legend_side="right")
  grDevices::dev.off()

  draw(ht, heatmap_legend_side="left", annotation_legend_side="right")
  invisible(list(mean_matrix=mat_mean, z_matrix=mat_z, missing_metabolites=met_missing))
}

glycolysis_result <- plot_pathway_heatmap(
  glycolysis_mets, "Glycolysis",
  file.path(output_dir,"PIVC1_Glycolysis_heatmap.pdf"), 7.0, 4.8
)

oxphos_tca_result <- plot_pathway_heatmap(
  oxphos_tca_mets, "OxPhos / TCA",
  file.path(output_dir,"PIVC1_OxPhos_TCA_heatmap.pdf"), 7.0, 5.2
)
