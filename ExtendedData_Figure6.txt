select_age_signature <- function(
  train_ids,
  n_each = 250
) {
  
  train_meta <- sample_info[
    match(
      train_ids,
      sample_info$sample_id
    ),
    ,
    drop = FALSE
  ]
  
  train_counts <- pb_counts_filtered[
    ,
    train_ids,
    drop = FALSE
  ]
  
  dge_train <- edgeR::DGEList(
    counts = train_counts
  )
  
  dge_train <- edgeR::calcNormFactors(
    dge_train,
    method = "TMM"
  )
  
  design_train <- model.matrix(
    ~ stage_num,
    data = train_meta
  )
  
  colnames(design_train) <- c(
    "Intercept",
    "Age"
  )
  
  voom_train <- limma::voom(
    dge_train,
    design_train,
    plot = FALSE
  )
  
  fit_train <- limma::lmFit(
    voom_train,
    design_train
  )
  
  fit_train <- limma::eBayes(
    fit_train,
    robust = TRUE
  )
  
  gene_stats <- limma::topTable(
    fit_train,
    coef = "Age",
    number = Inf,
    sort.by = "none"
  ) %>%
    tibble::rownames_to_column("gene")
  
  stage_means <- sapply(
    c(9, 10, 11, 12),
    function(current_stage) {
      
      rowMeans(
        voom_train$E[
          ,
          train_meta$stage_num ==
            current_stage,
          drop = FALSE
        ]
      )
    }
  )
  
  colnames(stage_means) <- c(
    "E9",
    "E10",
    "E11",
    "E12"
  )
  
  stage_means <- as.data.frame(
    stage_means
  ) %>%
    tibble::rownames_to_column("gene")
  
  candidates <- gene_stats %>%
    dplyr::transmute(
      gene,
      slope = logFC,
      t,
      FDR = adj.P.Val
    ) %>%
    dplyr::left_join(
      stage_means,
      by = "gene"
    ) %>%
    dplyr::mutate(
      trend = dplyr::case_when(
        slope > 0 &
          E9 < E10 &
          E10 < E11 &
          E11 < E12 ~
          "Increasing",
        
        slope < 0 &
          E9 > E10 &
          E10 > E11 &
          E11 > E12 ~
          "Decreasing",
        
        TRUE ~ "Non-monotonic"
      ),
      
      excluded = grepl(
        pattern = paste0(
          "^(mt-|Mt-|Rpl|Rps|",
          "Mrpl|Mrps|Hba|Hbb|Gm[0-9])"
        ),
        x = gene
      ) |
        gene %in% c("Malat1")
    ) %>%
    dplyr::filter(
      FDR < 0.01,
      abs(slope) >= 0.25,
      trend != "Non-monotonic",
      excluded == FALSE
    )
  
  increasing_genes <- candidates %>%
    dplyr::filter(
      trend == "Increasing"
    ) %>%
    dplyr::arrange(
      dplyr::desc(t)
    ) %>%
    dplyr::slice_head(
      n = n_each
    ) %>%
    dplyr::pull(gene)
  
  decreasing_genes <- candidates %>%
    dplyr::filter(
      trend == "Decreasing"
    ) %>%
    dplyr::arrange(t) %>%
    dplyr::slice_head(
      n = n_each
    ) %>%
    dplyr::pull(gene)
  
  return(
    list(
      increasing = increasing_genes,
      decreasing = decreasing_genes
    )
  )
}

#========================
# 26. 计算发育年龄评分
#========================

calculate_age_score <- function(
  signature,
  reference_ids,
  target_ids
) {
  
  stopifnot(length(signature$increasing)>0,length(signature$decreasing)>0)
  selected_genes <- unique(
    c(
      signature$increasing,
      signature$decreasing
    )
  )
  
  reference_expression <- logCPM_all[
    selected_genes,
    reference_ids,
    drop = FALSE
  ]
  
  reference_mean <- rowMeans(
    reference_expression
  )
  
  reference_sd <- apply(
    reference_expression,
    1,
    sd
  )
  
  valid_genes <- names(
    reference_sd[
      is.finite(reference_sd) &
        reference_sd > 0
    ]
  )
  
  increasing_valid <- intersect(
    signature$increasing,
    valid_genes
  )
  
  decreasing_valid <- intersect(
    signature$decreasing,
    valid_genes
  )
  
  stopifnot(length(increasing_valid)>0,length(decreasing_valid)>0)
  target_expression <- logCPM_all[
    valid_genes,
    target_ids,
    drop = FALSE
  ]
  
  target_z <- sweep(
    target_expression,
    1,
    reference_mean[valid_genes],
    "-"
  )
  
  target_z <- sweep(
    target_z,
    1,
    reference_sd[valid_genes],
    "/"
  )
  
  increasing_score <- colMeans(
    target_z[
      increasing_valid,
      ,
      drop = FALSE
    ]
  )
  
  decreasing_score <- colMeans(
    target_z[
      decreasing_valid,
      ,
      drop = FALSE
    ]
  )
  
  data.frame(
    sample_id = target_ids,
    age_score =
      increasing_score -
      decreasing_score
  )
}

cv_results_list <- list()

cv_folds <- c("E1", "E2", "E3")

for (current_fold in cv_folds) {
  
  test_ids <- vivo_cv_meta %>%
    dplyr::filter(
      cv_fold == current_fold
    ) %>%
    dplyr::pull(sample_id)
  
  train_ids <- vivo_cv_meta %>%
    dplyr::filter(
      cv_fold != current_fold
    ) %>%
    dplyr::pull(sample_id)
  
  current_signature <- select_age_signature(
    train_ids = train_ids,
    n_each = 250
  )
  
  train_scores <- calculate_age_score(
    signature = current_signature,
    reference_ids = train_ids,
    target_ids = train_ids
  )
  
  test_scores <- calculate_age_score(
    signature = current_signature,
    reference_ids = train_ids,
    target_ids = test_ids
  )
  
  train_model_df <- train_scores %>%
    dplyr::left_join(
      vivo_cv_meta %>%
        dplyr::select(
          sample_id,
          stage_num
        ),
      by = "sample_id"
    )
  
  calibration_model <- lm(
    stage_num ~ age_score,
    data = train_model_df
  )
  
  test_scores$predicted_age <- predict(
    calibration_model,
    newdata = test_scores
  )
  
  test_scores <- test_scores %>%
    dplyr::left_join(
      vivo_cv_meta %>%
        dplyr::select(
          sample_id,
          stage_num
        ),
      by = "sample_id"
    ) %>%
    dplyr::mutate(
      cv_fold = current_fold,
      n_increasing = length(
        current_signature$increasing
      ),
      n_decreasing = length(
        current_signature$decreasing
      )
    )
  
  cv_results_list[[current_fold]] <- test_scores
}

# Shared f/h calculation from Figure3H: expression fraction and scaled mean.
# RNA data must contain log1p-normalized expression; pooled cells are used.
marker_dot <- function(obj,subtype_col,subtype_levels,genes,prefix) {
  obj <- prepare_meta(obj); DefaultAssay(obj) <- "RNA"
  stopifnot(subtype_col %in% names(obj[[]]),!anyNA(obj[[]][[subtype_col]]),
            all(as.character(obj[[]][[subtype_col]]) %in% subtype_levels))
  found <- intersect(genes,rownames(obj)); missing <- setdiff(genes,rownames(obj))
  writeLines(missing,file.path(out_dir,paste0(prefix,"_missing_markers.txt")))
  if(length(missing)) warning(prefix,": missing markers: ",paste(missing,collapse=", "))
  if(!length(found)) stop("No marker genes matched: ",prefix)
  df <- FetchData(obj,vars=c("sample2",subtype_col,found),slot="data")
  names(df)[names(df)==subtype_col] <- "Subtype"
  z <- df %>% pivot_longer(cols=all_of(found),names_to="Gene",values_to="expr") %>%
    group_by(sample2,Subtype,Gene) %>% summarise(pct.exp=mean(expr>0)*100,
      avg.exp=mean(expm1(expr)),.groups="drop") %>%
    group_by(Gene) %>% mutate(avg.scaled=as.numeric(scale(log1p(avg.exp)))) %>% ungroup()
  z$avg.scaled[!is.finite(z$avg.scaled)] <- 0
  z$avg.scaled <- pmax(pmin(z$avg.scaled,1.5),-1.5)
  z$Gene <- factor(z$Gene,levels=found)
  z$Subtype <- factor(z$Subtype,levels=rev(subtype_levels))
  p <- ggplot(z,aes(Gene,Subtype,size=pct.exp,color=avg.scaled))+geom_point()+
    facet_grid(sample2~.,labeller=labeller(sample2=culture_labels))+
    scale_color_gradient2(low="#2C7FB8",mid="white",high="#D73027",limits=c(-1.5,1.5),name="Scaled expression")+
    scale_size_continuous(range=c(0,5),limits=c(0,100),breaks=c(0,25,50,75,100),name="Percent expressed")+
    labs(x=NULL,y=NULL)+theme_bw(base_size=10)+
    theme(axis.text.x=element_text(angle=60,hjust=1,face="italic"),panel.grid=element_blank())
  save_plot(p,prefix,12,8)
  write.csv(z,file.path(out_dir,paste0(prefix,"_data.csv")),row.names=FALSE)
  p
}
# Shared g/i: denominator is all cells in the corresponding subtype object.
composition <- function(obj,col,levels_sub,cols,prefix) {
  obj <- prepare_meta(obj)
  meta <- obj[[]]
  stopifnot(col %in% names(meta),!anyNA(meta[[col]]),all(as.character(meta[[col]]) %in% levels_sub))
  meta$Subtype <- factor(meta[[col]],levels=levels_sub)
  z <- meta %>% count(sample2,sample1,Subtype,name="n",.drop=TRUE) %>%
    group_by(sample2,sample1) %>% mutate(total=sum(n),prop=n/total) %>% ungroup()
  p <- ggplot(z,aes(sample1,prop,fill=Subtype))+geom_col(width=.82,color="grey25",linewidth=.25)+
    facet_grid(.~sample2,scales="free_x",space="free_x",labeller=labeller(sample2=culture_labels))+
    scale_x_discrete(labels=stage_labels)+scale_y_continuous(labels=scales::percent_format(),breaks=seq(0,1,.25),expand=expansion(mult=c(0,.02)))+
    scale_fill_manual(values=cols,breaks=levels_sub,drop=FALSE)+
    labs(x=NULL,y="Cell proportion",fill=NULL)+theme_classic(base_size=11)+
    theme(axis.text.x=element_text(angle=60,hjust=1),strip.background=element_blank())
  save_plot(p,prefix,11,4)
  write.csv(z,file.path(out_dir,paste0(prefix,"_data.csv")),row.names=FALSE)
  p
}
