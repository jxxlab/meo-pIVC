# ============================================================
# Strict pseudo-bulk correlation analysis
# meo-pIVC2 (D1-D6) / meo-pIVC1 (RD1-RD4) vs in vivo E9-E12
#
# Input:
#   moe-E9-D6-RD4-373319.rds
#
# Method:
#   1. Explicitly define all 42 biological replicates
#   2. Strictly verify replicate names and cell numbers
#   3. Sum raw UMI counts within EACH biological replicate
#   4. edgeR filterByExpr
#   5. TMM normalization
#   6. log2-CPM transformation
#   7. Average 3 biological replicates within each stage/day
#   8. Pearson correlation across retained genes
#   9. Plot D1-D6 / RD1-RD4 versus E9-E12
#
# NOTE:
# The code assumes the replicate IDs (E9E1, E9E2, ..., D6E3)
# are stored in pbmc@meta.data$orig.ident.
# If they are stored in another metadata column, ONLY change:
#     replicate_col <- "orig.ident"
# ============================================================


# ============================================================
# 0. Load packages
# ============================================================

library(Seurat)
library(Matrix)
library(edgeR)
library(dplyr)
library(tidyr)
library(ggplot2)
library(patchwork)
library(scales)


# ============================================================
# 1. Read Seurat object
# ============================================================

pbmc <- readRDS("moe-E9-D6-RD4-373319.rds")

DefaultAssay(pbmc) <- "RNA"

# Stage/day metadata
Idents(pbmc) <- "sample1"

cat("Seurat object:\n")
print(pbmc)

cat("\nCell numbers by sample1:\n")
print(table(pbmc$sample1))

cat("\nMetadata columns:\n")
print(names(pbmc@meta.data))


# ============================================================
# 2. Explicitly specify the biological-replicate metadata column
# ============================================================

replicate_col <- "orig.ident"

if (!replicate_col %in% colnames(pbmc@meta.data)) {
  stop(
    paste0(
      "Metadata column '", replicate_col, "' was not found.\n",
      "Please change replicate_col to the column containing ",
      "E9E1, E9E2, ..., D6E3."
    )
  )
}

  "D5E1",  "D5E2",  "D5E3",
  "D6E1",  "D6E2",  "D6E3"

# ============================================================

  E9E2   = "E9",
  E9E3   = "E9",

  E10E1  = "E10",
  E10E2  = "E10",

stage_order <- c(
  "E9", "E10", "E11", "E12",
  "RD1", "RD2", "RD3", "RD4",
  "D1", "D2", "D3", "D4", "D5", "D6"
)


# ============================================================
# 6. Strict metadata checks
# ============================================================

meta <- pbmc@meta.data

# Seurat metadata row names must match Seurat cell barcodes
if (!identical(rownames(meta), colnames(pbmc))) {
  stop(
    "rownames(pbmc@meta.data) are not identical to colnames(pbmc). ",
    "Do not continue until the metadata/cell order is resolved."
  )
}

replicate_id <- as.character(meta[[replicate_col]])
names(replicate_id) <- rownames(meta)

if (any(is.na(replicate_id))) {
  stop("NA values were found in the biological-replicate metadata column.")
}

observed_replicates <- unique(replicate_id)

missing_replicates <- setdiff(
  expected_replicates,
  observed_replicates
)

unexpected_replicates <- setdiff(
  observed_replicates,
  expected_replicates
)

if (length(missing_replicates) > 0) {
  stop(
    paste(
      "Missing expected biological replicates:",
      paste(missing_replicates, collapse = ", ")
    )
  )
}

if (length(unexpected_replicates) > 0) {
  stop(
    paste(
      "Unexpected biological replicate names were found:",
      paste(unexpected_replicates, collapse = ", ")
    )
  )
}


# ============================================================
# 7. Strictly verify cell number for every biological replicate
# ============================================================

actual_ncells_table <- table(
  factor(
    replicate_id,
    levels = expected_replicates
  )
)

actual_ncells <- as.integer(actual_ncells_table)
names(actual_ncells) <- expected_replicates

sample_qc <- data.frame(
  replicate = expected_replicates,
  stage = unname(replicate_to_stage[expected_replicates]),
  expected_cells = as.integer(expected_ncells),
  actual_cells = as.integer(actual_ncells),
  difference = as.integer(actual_ncells - expected_ncells),
  stringsAsFactors = FALSE
)

cat("\nStrict biological-replicate cell-count check:\n")
print(sample_qc)

write.csv(
  sample_qc,
  "pseudo_bulk_sample_cell_count_QC.csv",
  row.names = FALSE,
  quote = FALSE
)

if (any(sample_qc$difference != 0)) {
  stop(
    paste0(
      "Cell-number QC failed. ",
      "At least one biological replicate does not match the expected cell number. ",
      "See pseudo_bulk_sample_cell_count_QC.csv."
    )
  )
}

if (sum(actual_ncells) != 373319) {
  stop(
    paste0(
      "Total cell count is ", sum(actual_ncells),
      ", but 373319 was expected."
    )
  )
}

cat("\nCell-number QC PASSED.\n")
cat("All 42 biological replicates exactly match the expected counts.\n")
cat("Total cells = ", sum(actual_ncells), "\n", sep = "")


# ============================================================
# 8. Strictly verify sample1 stage/day labels
# ============================================================

if (!"sample1" %in% colnames(meta)) {
  stop("Metadata column 'sample1' was not found.")
}

expected_stage_per_cell <- unname(
  replicate_to_stage[replicate_id]
)

observed_stage_per_cell <- as.character(
  meta$sample1
)

stage_mismatch <- which(
  observed_stage_per_cell != expected_stage_per_cell
)

if (length(stage_mismatch) > 0) {

  mismatch_examples <- data.frame(
    cell = rownames(meta)[stage_mismatch],
    replicate = replicate_id[stage_mismatch],
    expected_stage = expected_stage_per_cell[stage_mismatch],
    sample1 = observed_stage_per_cell[stage_mismatch]
  )

  write.csv(
    mismatch_examples,
    "sample1_stage_mismatch.csv",
    row.names = FALSE,
    quote = FALSE
  )

  stop(
    paste0(
      "sample1 QC failed: ",
      length(stage_mismatch),
      " cells have inconsistent stage/day labels. ",
      "See sample1_stage_mismatch.csv."
    )
  )
}

cat("\nsample1 stage/day QC PASSED.\n")


# ============================================================
# 9. Extract RAW RNA UMI counts
# ============================================================

counts <- GetAssayData(
  pbmc,
  assay = "RNA",
  slot = "counts"
)

if (!identical(colnames(counts), rownames(meta))) {
  stop(
    "The order of cells in the RNA count matrix does not match metadata."
  )
}

cat("\nRaw RNA count matrix:\n")
print(dim(counts))


# ============================================================
# 10. Construct pseudo-bulk counts for EACH biological replicate
# ============================================================
# Raw UMI counts from all cells belonging to the same biological
# replicate are summed. This gives a genes x 42 samples matrix.

cell_replicate <- factor(
  replicate_id,
  levels = expected_replicates
)

design_pb <- Matrix::sparse.model.matrix(
  ~ 0 + cell_replicate
)

colnames(design_pb) <- expected_replicates

pseudo_counts <- counts %*% design_pb

rownames(pseudo_counts) <- rownames(counts)
colnames(pseudo_counts) <- expected_replicates

if (!identical(
  colnames(pseudo_counts),
  expected_replicates
)) {
  stop("Pseudo-bulk sample order is incorrect.")
}

cat("\nPseudo-bulk raw count matrix:\n")
print(dim(pseudo_counts))

cat("\nPseudo-bulk library sizes:\n")
print(colSums(pseudo_counts))


# ============================================================
# 11. Define stage/day for each of the 42 pseudo-bulk samples
# ============================================================

rep_stage <- factor(
  unname(
    replicate_to_stage[
      colnames(pseudo_counts)
    ]
  ),
  levels = stage_order
)

replicate_annotation <- data.frame(
  replicate = colnames(pseudo_counts),
  stage = as.character(rep_stage),
  cells = actual_ncells[colnames(pseudo_counts)],
  library_size = as.numeric(colSums(pseudo_counts)),
  stringsAsFactors = FALSE
)

write.csv(
  replicate_annotation,
  "pseudo_bulk_replicate_annotation.csv",
  row.names = FALSE,
  quote = FALSE
)

cat("\nPseudo-bulk replicate annotation:\n")
print(replicate_annotation)


# ============================================================
# 12. edgeR expression filtering
# ============================================================
# Genes with insufficient expression across biological replicates
# are removed before correlation analysis.

y <- edgeR::DGEList(
  counts = pseudo_counts,
  group = rep_stage
)

keep <- edgeR::filterByExpr(
  y,
  group = rep_stage
)

cat("\nGenes before filterByExpr: ",
    nrow(y), "\n", sep = "")

cat("Genes retained after filterByExpr: ",
    sum(keep), "\n", sep = "")

cat("Genes removed by filterByExpr: ",
    sum(!keep), "\n", sep = "")

y <- y[
  keep,
  ,
  keep.lib.sizes = FALSE
]


# ============================================================
# 13. TMM normalization
# ============================================================

y <- edgeR::calcNormFactors(
  y,
  method = "TMM"
)

cat("\nedgeR library sizes and TMM normalization factors:\n")
print(y$samples)

write.csv(
  y$samples,
  "pseudo_bulk_TMM_normalization_factors.csv",
  quote = FALSE
)


# ============================================================
# 14. Calculate replicate-level log2-CPM
# ============================================================

logcpm_rep <- edgeR::cpm(
  y,
  log = TRUE,
  prior.count = 1
)

if (any(!is.finite(logcpm_rep))) {
  stop("Non-finite values were found in replicate-level logCPM.")
}

cat("\nReplicate-level logCPM matrix:\n")
print(dim(logcpm_rep))


# ============================================================
# 15. Average the 3 biological replicates within each stage/day
# ============================================================
# IMPORTANT:
# The mean is calculated AFTER:
#   replicate-level pseudo-bulk
#   expression filtering
#   TMM normalization
#   logCPM transformation

stage_mean_logcpm <- matrix(
  NA_real_,
  nrow = nrow(logcpm_rep),
  ncol = length(stage_order),
  dimnames = list(
    rownames(logcpm_rep),
    stage_order
  )
)

for (st in stage_order) {

  replicate_names_for_stage <- names(
    replicate_to_stage
  )[
    replicate_to_stage == st
  ]

  if (length(replicate_names_for_stage) != 3) {
    stop(
      paste0(
        "Stage ", st,
        " does not have exactly 3 explicitly defined biological replicates."
      )
    )
  }

  if (!all(
    replicate_names_for_stage %in%
      colnames(logcpm_rep)
  )) {
    stop(
      paste0(
        "Not all biological replicates for ", st,
        " are present in logCPM matrix."
      )
    )
  }

  stage_mean_logcpm[, st] <- rowMeans(
    logcpm_rep[
      ,
      replicate_names_for_stage,
      drop = FALSE
    ]
  )
}

if (any(!is.finite(stage_mean_logcpm))) {
  stop("Non-finite values were found in stage-level mean logCPM.")
}

cat("\nStage/day mean logCPM matrix:\n")
print(dim(stage_mean_logcpm))


# ============================================================
# 16. Pearson correlation
# ============================================================
# Correlation is calculated across all genes retained by
# filterByExpr.

cor_mat <- cor(
  stage_mean_logcpm,
  method = "pearson",
  use = "everything"
)

cat("\nFull Pearson correlation matrix:\n")
print(round(cor_mat, 3))


# ============================================================
# 17. Extract figure matrices
# ============================================================

in_vivo_stages <- c(
  "E9", "E10", "E11", "E12"
)

meo_pIVC2_days <- c(

  in_vivo_stages,
  drop = FALSE
]

cor_RD <- cor_mat[
  meo_pIVC1_days,
  in_vivo_stages,
  drop = FALSE
]

cat("\nmeo-pIVC2: D1-D6 vs E9-E12\n")
print(round(cor_D, 2))

cat("\nmeo-pIVC1: RD1-RD4 vs E9-E12\n")
print(round(cor_RD, 2))
