# ======================================================================
# Figure 3c / Figure 3i
# Reusable STRICT pseudo-bulk Pearson-correlation workflow
#
# Method:
#   1. biological replicate = orig.ident
#   2. sum RAW UMI counts within each biological replicate
#   3. edgeR filterByExpr
#   4. TMM normalization
#   5. log2-CPM
#   6. average the 3 replicates within each stage/day
#   7. Pearson correlation across retained genes
#
# Figure 3c:
#   full dataset; D1-D6 and RD1-RD4 versus E9-E12
#
# Figure 3i:
#   same method applied to the gut/endoderm subset;
#   plot D1-D6 versus E9-E12
# ======================================================================

run_strict_pseudobulk_correlation <- function(
  obj,
  prefix,
  replicate_col = "orig.ident",
  expected_ncells = NULL,
  plot_pivc1 = TRUE
) {

  DefaultAssay(obj) <- "RNA"
  meta <- obj@meta.data

  # ----------------------------------------------------------
  # A. strict metadata checks
  # ----------------------------------------------------------
  stopifnot(replicate_col %in% colnames(meta))
  stopifnot("sample1" %in% colnames(meta))
  stopifnot(identical(rownames(meta), colnames(obj)))

  replicate_id <- as.character(meta[[replicate_col]])
  names(replicate_id) <- rownames(meta)

  observed_reps <- unique(replicate_id)

  missing_reps <- setdiff(expected_replicates, observed_reps)
  extra_reps   <- setdiff(observed_reps, expected_replicates)

  if (length(missing_reps) > 0) {
    stop("Missing replicates: ", paste(missing_reps, collapse = ", "))
  }

  if (length(extra_reps) > 0) {
    stop("Unexpected replicates: ", paste(extra_reps, collapse = ", "))
  }

  # sample1 must agree with explicit replicate -> stage mapping
  expected_stage_per_cell <- unname(replicate_to_stage[replicate_id])
  observed_stage_per_cell <- as.character(meta$sample1)

  if (any(observed_stage_per_cell != expected_stage_per_cell)) {
    stop("sample1 does not agree with explicit replicate_to_stage mapping.")
  }

  # Optional exact cell-count QC.
  # Use this for Figure 3c full 373,319-cell object.
  actual_ncells <- table(
    factor(replicate_id, levels = expected_replicates)
  )

  if (!is.null(expected_ncells)) {
    expected_ncells <- expected_ncells[expected_replicates]

    qc <- data.frame(
      replicate = expected_replicates,
      expected_cells = as.integer(expected_ncells),
      actual_cells = as.integer(actual_ncells),
      difference = as.integer(actual_ncells) - as.integer(expected_ncells)
    )

    write.csv(
      qc,
      paste0(prefix, "_sample_cell_count_QC.csv"),
      row.names = FALSE
    )

    if (any(qc$difference != 0)) {
      stop("Exact cell-count QC failed for ", prefix)
    }
  }

  # ----------------------------------------------------------
  # B. raw count extraction
  # ----------------------------------------------------------
  counts <- GetAssayData(
    obj,
    assay = "RNA",
    slot = "counts"
  )

  stopifnot(identical(colnames(counts), rownames(meta)))

  # ----------------------------------------------------------
  # C. replicate-level pseudo-bulk
  # ----------------------------------------------------------
  cell_replicate <- factor(
    replicate_id,
    levels = expected_replicates
  )

  design_pb <- Matrix::sparse.model.matrix(
    ~ 0 + cell_replicate
  )

  colnames(design_pb) <- expected_replicates

  pseudo_counts <- counts %*% design_pb
  rownames(pseudo_counts) <- rownames(counts)
  colnames(pseudo_counts) <- expected_replicates

  # ----------------------------------------------------------
  # D. edgeR filtering + TMM + log2-CPM
  # ----------------------------------------------------------
  rep_stage <- factor(
    unname(replicate_to_stage[colnames(pseudo_counts)]),
    levels = stage_order
  )

  y <- edgeR::DGEList(
    counts = pseudo_counts,
    group = rep_stage
  )

  keep <- edgeR::filterByExpr(
    y,
    group = rep_stage
  )

  y <- y[
    keep,
    ,
    keep.lib.sizes = FALSE
  ]

  y <- edgeR::calcNormFactors(
    y,
    method = "TMM"
  )

  logcpm_rep <- edgeR::cpm(
    y,
    log = TRUE,
    prior.count = 1
  )

  # ----------------------------------------------------------
  # E. average exactly 3 biological replicates per stage/day
  # ----------------------------------------------------------
  stage_mean_logcpm <- matrix(
    NA_real_,
    nrow = nrow(logcpm_rep),
    ncol = length(stage_order),
    dimnames = list(
      rownames(logcpm_rep),
      stage_order
    )
  )

  for (st in stage_order) {

    reps_st <- names(replicate_to_stage)[
      replicate_to_stage == st
    ]

    if (length(reps_st) != 3) {
      stop(st, " does not have exactly 3 explicitly defined replicates.")
    }

    stage_mean_logcpm[, st] <- rowMeans(
      logcpm_rep[, reps_st, drop = FALSE]
    )
  }

  # ----------------------------------------------------------
  # F. Pearson correlation across retained genes
  # ----------------------------------------------------------
  cor_mat <- cor(
    stage_mean_logcpm,
    method = "pearson",
    use = "everything"
  )

  vivo <- c("E9","E10","E11","E12")
  p2   <- c("D1","D2","D3","D4","D5","D6")
  p1   <- c("RD1","RD2","RD3","RD4")

  cor_D  <- cor_mat[p2, vivo, drop = FALSE]
  cor_RD <- cor_mat[p1, vivo, drop = FALSE]

  write.csv(
    cor_mat,
    paste0(prefix, "_full_Pearson.csv")
  )

  write.csv(
    cor_D,
    paste0(prefix, "_meo_pIVC2_vs_in_vivo.csv")
  )

  write.csv(
    cor_RD,
    paste0(prefix, "_meo_pIVC1_vs_in_vivo.csv")
  )

  write.table(
    rownames(stage_mean_logcpm),
    paste0(prefix, "_genes_used_for_Pearson.txt"),
    quote = FALSE,
    row.names = FALSE,
    col.names = FALSE
  )

  # ----------------------------------------------------------
  # G. plotting helper
  # ----------------------------------------------------------
  make_cor_df <- function(mat, day_prefix = "Day") {
    z <- as.data.frame(mat, check.names = FALSE) %>%
      mutate(CultureDay = rownames(.)) %>%
      pivot_longer(
        cols = all_of(vivo),
        names_to = "Stage",
        values_to = "Correlation"
      )

    # Extract final number from D1/RD1 etc.
    z$Day <- paste(
      day_prefix,
      gsub("[^0-9]", "", z$CultureDay)
    )

    z$Stage <- factor(z$Stage, levels = vivo)
    z
  }

  D_df <- make_cor_df(cor_D)
  D_df$Day <- factor(
    D_df$Day,
    levels = rev(paste("Day", 1:6))
  )

  RD_df <- make_cor_df(cor_RD)
  RD_df$Day <- factor(
    RD_df$Day,
    levels = rev(paste("Day", 1:4))
  )

  cor_scale <- scale_fill_gradientn(
    colours = c(
      "#2C7BB6",
      "#ABD9E9",
      "#FFFFBF",
      "#FDAE61",
      "#D7191C"
    ),
    limits = c(0.4, 1.0),
    breaks = c(0.4, 0.6, 0.8, 1.0),
    oob = scales::squish,
    name = "Correlation"
  )

  p_D <- ggplot(
    D_df,
    aes(x = Stage, y = Day, fill = Correlation)
  ) +
    geom_tile(color = "grey70", linewidth = 0.45) +
    geom_text(
      aes(label = sprintf("%.2f", Correlation)),
      size = 4.2
    ) +
    cor_scale +
    coord_fixed() +
    labs(x = NULL, y = "meo-pIVC2") +
    theme_classic(base_size = 12) +
    theme(
      axis.ticks = element_blank(),
      axis.line = element_blank()
    )

  if (plot_pivc1) {

    p_RD <- ggplot(
      RD_df,
      aes(x = Stage, y = Day, fill = Correlation)
    ) +
      geom_tile(color = "grey70", linewidth = 0.45) +
      geom_text(
        aes(label = sprintf("%.2f", Correlation)),
        size = 4.2
      ) +
      cor_scale +
      coord_fixed() +
      labs(x = NULL, y = "meo-pIVC1") +
      theme_classic(base_size = 12) +
      theme(
        axis.ticks = element_blank(),
        axis.line = element_blank(),
        legend.position = "none"
      )

    p_final <- p_D / p_RD +
      patchwork::plot_layout(
        heights = c(6, 4)
      )

    ggsave(
      paste0(prefix, "_Pearson_heatmap.pdf"),
      p_final,
      width = 4.6,
      height = 6.0
    )

  } else {

    ggsave(
      paste0(prefix, "_Pearson_heatmap.pdf"),
      p_D,
      width = 4.6,
      height = 4.0
    )

    p_final <- p_D
  }

  return(
    list(
      pseudo_counts = pseudo_counts,
      logcpm_rep = logcpm_rep,
      stage_mean_logcpm = stage_mean_logcpm,
      correlation = cor_mat,
      cor_pIVC2 = cor_D,
      cor_pIVC1 = cor_RD,
      plot = p_final
    )
  )
}

def prepare_cx1_palette(adata):

    adata.obs["CX1"] = adata.obs["CX1"].astype(str)

    cx1_categories = sorted(
        adata.obs["CX1"].unique(),
        key=lambda z: int(z)
    )

    adata.obs["CX1"] = pd.Categorical(
        adata.obs["CX1"],
        categories=cx1_categories,
        ordered=True
    )

    full_palette = {
        str(i + 1): color.upper()
        for i, color in enumerate(x_colors)
    }

    return {
        cluster: full_palette[cluster]
        for cluster in cx1_categories
    }


def run_velocity_for_group(adata_all, group_name):

    # Split into Vivo / Vitro1 / Vitro2 FIRST,
    # then recalculate graph/moments/velocity for that group.
    adata = adata_all[
        adata_all.obs["sample2"].isin([group_name])
    ].copy()

    print(group_name, adata.shape)

    scv.pp.filter_and_normalize(
        adata,
        min_shared_counts=20,
        n_top_genes=2000
    )

    scv.pp.neighbors(
        adata,
        n_pcs=30,
        n_neighbors=30
    )

    scv.pp.moments(
        adata,
        n_pcs=30,
        n_neighbors=30
    )

    scv.tl.velocity(adata)

    scv.tl.velocity_graph(
        adata,
        n_jobs=10
    )

    return adata


def plot_velocity_group(
    adata,
    x_palette,
    group_name,
    output_prefix
):

    fig, ax = plt.subplots(figsize=(4, 4))

    scv.pl.velocity_embedding_stream(
        adata,
        basis="umap",
        color="CX1",
        palette=x_palette,
        size=10,
        alpha=0.5,
        density=0.6,
        arrow_size=1,
        linewidth=1.0,
        smooth=0.8,
        max_length=6,
        legend_loc="on data",
        legend_fontsize=6,
        legend_fontweight="bold",
        title="",
        ax=ax,
        show=False
    )

    # Keep labels editable
    for txt in ax.texts:
        txt.set_path_effects([])
        txt.set_rasterized(False)
        txt.set_fontsize(6)
        txt.set_fontweight("bold")
        txt.set_color("black")

    # Prevent non-finite linewidth errors
    for i, artist in enumerate(ax.collections):

        if hasattr(artist, "get_linewidths"):

            lw = np.asarray(
                artist.get_linewidths(),
                dtype=float
            )

            if lw.size > 0:

                n_bad = np.sum(~np.isfinite(lw))

                if n_bad > 0:

                    lw = np.nan_to_num(
                        lw,
                        nan=0.0,
                        posinf=0.0,
                        neginf=0.0
                    )

                    artist.set_linewidths(lw)

    # Rasterize cells only; keep text/streamlines editable
    for artist in ax.collections:
        if isinstance(artist, PathCollection):
            artist.set_rasterized(True)

    output_file = (
        f"{output_prefix}_{group_name}_editable.pdf"
    )

    fig.savefig(
        output_file,
        format="pdf",
        dpi=600,
        bbox_inches="tight",
        facecolor="white"
    )

    plt.show()
    plt.close(fig)


def make_velocity_panel(
    input_h5ad,
    output_prefix
):

    adata = scv.read(input_h5ad)

    print(adata)

    # Initial whole-subset processing
    scv.pp.filter_and_normalize(
        adata,
        min_shared_counts=20,
        n_top_genes=2000
    )

    scv.pp.moments(
        adata,
        n_pcs=30,
        n_neighbors=30
    )

    scv.tl.velocity(adata)

    scv.tl.velocity_graph(
        adata,
        n_jobs=10
    )

    x_palette = prepare_cx1_palette(adata)
    adata_all = adata.copy()

    for group_name in [
        "Vivo",
        "Vitro1",
        "Vitro2"
    ]:

        adata_group = run_velocity_for_group(
            adata_all,
            group_name
        )

        plot_velocity_group(
            adata_group,
            x_palette,
            group_name,
            output_prefix
        )
