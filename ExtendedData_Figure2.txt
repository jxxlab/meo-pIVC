# ============================================================
# PIVC1 PCA of embryo samples
# Figure-only standalone script
#
# R version: 4.1.3
# Packages: readxl 1.4.1; dplyr 1.2.1; stringr 1.5.0;
#           ggplot2 3.4.0; ggforce
#

suppressPackageStartupMessages({
  library(readxl); library(dplyr); library(stringr)
  library(ggplot2); library(ggforce)
})

input_file <- "data/all_sample_data_tm36.xlsx"
output_dir <- "results"
dir.create(output_dir, recursive=TRUE, showWarnings=FALSE)

culture_colors <- c("In vivo"="#1b9e77", "In vitro"="#d95f02")
stage_order <- c("E11","E12","D11","D12")

# ------------------------------------------------------------
# Load PIVC1 embryo data
# ------------------------------------------------------------

dat <- read_excel(input_file) %>% as.data.frame(check.names=FALSE)
if(!"Compounds" %in% colnames(dat)) stop("Column 'Compounds' was not found.")

sample_pattern <- paste0("^(", paste(stage_order, collapse="|"), ")E[0-9]+$")
sample_cols <- colnames(dat)[str_detect(toupper(colnames(dat)), sample_pattern)]
if(length(sample_cols)==0) stop("No PIVC1 embryo sample columns were detected.")

stage_regex <- paste0("^(", paste(stage_order[order(nchar(stage_order), decreasing=TRUE)], collapse="|"), ")")

sample_info <- data.frame(Sample=sample_cols, stringsAsFactors=FALSE) %>%
  mutate(
    Culture=ifelse(str_starts(toupper(Sample),"E"), "In vivo", "In vitro"),
    Stage=str_extract(toupper(Sample), stage_regex),
    Replicate=as.integer(str_extract(Sample, "[0-9]+$"))
  ) %>%
  mutate(
    Culture=factor(Culture, levels=c("In vivo","In vitro")),
    Stage=factor(Stage, levels=stage_order)
  ) %>%
  arrange(Culture, Stage, Replicate)

mat <- dat %>% select(all_of(sample_info$Sample)) %>% as.matrix()
storage.mode(mat) <- "numeric"

mat_log <- log2(mat + 1)

  stop("Non-finite values were detected after log2(x+1) transformation.")
}

# PCA
# ------------------------------------------------------------

p <- ggplot(pca_df, aes(PC1, PC2, color=Culture)) +

    aes(fill=Culture, group=Culture),

  geom_point(size=3) +
  geom_text(aes(label=Sample), size=5, vjust=-0.8, show.legend=FALSE) +
  scale_color_manual(values=culture_colors) +
  scale_fill_manual(values=culture_colors) +
  scale_x_continuous(expand=expansion(mult=c(0.15,0.12))) +
  scale_y_continuous(expand=expansion(mult=c(0.35,0.12))) +
  labs(
    title="PCA of Embryo samples",
    x=paste0("PC1 (", pc_var[1], "%)"),
    y=paste0("PC2 (", pc_var[2], "%)")
  ) +
  theme_bw(base_size=14) +
  theme(
    panel.grid=element_blank(),
    panel.border=element_rect(color="black", fill=NA, size=1),
    plot.title=element_text(hjust=0.5, face="bold", size=20),
    axis.title=element_text(size=18),
    axis.text=element_text(size=15),
    legend.title=element_blank(),
    legend.text=element_text(size=20),
    aspect.ratio=1
  )

ggsave(
  filename=file.path(output_dir, "PIVC1_PCA_embryo_samples.pdf"),
  plot=p, width=7.2, height=7.2, units="in", device="pdf"
)

print(p)

# PANEL c: uploaded source preserved
if (RUN_C) {
# ============================================================
# PIVC1 global stage-resolved metabolic pathway dynamics
# Standalone figure-reproduction script
#
# R version: 4.1.3
# Packages: readxl 1.4.1; dplyr 1.2.1; stringr 1.5.0;
#           ggplot2 3.4.0; scales 1.3.0
#
# Analysis:
#   log2(x+1) -> biological-replicate stage mean
#   -> metabolite row-wise Z-score across E11/E12/D11/D12
#   -> pathway metabolites vs remaining KEGG-mapped metabolites
#   -> two-sided Wilcoxon rank-sum test
#   -> directional pathway score =
#      mean(pathway Z-score) - mean(background Z-score)
#
# Display:
#   non-core pathway: retained if raw P < 0.05 at any stage
#   core pathway: always retained
#   bubble size: -log10(raw P value)
#   bubble color: directional pathway score
# ============================================================

suppressPackageStartupMessages({
  library(readxl); library(dplyr); library(stringr)
  library(ggplot2); library(scales)
})

data_file <- "data/all_sample_data_tm36.xlsx"
kegg_file <- "data/kegg_compound_pathway.csv"
output_dir <- "results"
dir.create(output_dir, recursive=TRUE, showWarnings=FALSE)

stage_order <- c("E11","E12","D11","D12")
min_pathway_metabolites <- 3L

core_pathways <- c(
  "Butanoate metabolism","Citrate cycle (TCA cycle)","Oxidative phosphorylation",
  "Pyruvate metabolism","Pentose phosphate pathway","Glyoxylate and dicarboxylate metabolism",
  "Nicotinate and nicotinamide metabolism","Glycolysis / Gluconeogenesis",
  "Fatty acid degradation","Fructose and mannose metabolism","Propanoate metabolism"
)

category_order <- c(
  "Carbohydrate & energy metabolism","Amino acid & nucleotide metabolism",
  "Lipid metabolism","Cofactor & vitamin metabolism","Other metabolism"
)

category_labels <- c(
  "Carbohydrate & energy metabolism"="Carbohydrate & energy\nmetabolism",
  "Amino acid & nucleotide metabolism"="Amino acid & nucleotide\nmetabolism",
  "Lipid metabolism"="Lipid\nmetabolism",
  "Cofactor & vitamin metabolism"="Cofactor & vitamin\nmetabolism",
  "Other metabolism"="Other\nmetabolism"
)

assign_category <- function(x) {
  case_when(
    x %in% c(
      "Glycolysis / Gluconeogenesis","Citrate cycle (TCA cycle)","Pyruvate metabolism",
      "Pentose phosphate pathway","Oxidative phosphorylation","Fructose and mannose metabolism",
      "Galactose metabolism","Starch and sucrose metabolism","Propanoate metabolism",
      "Butanoate metabolism","Pentose and glucuronate interconversions",
      "Glyoxylate and dicarboxylate metabolism","Amino sugar and nucleotide sugar metabolism"
    ) ~ "Carbohydrate & energy metabolism",

    x %in% c(
      "Alanine, aspartate and glutamate metabolism","Arginine biosynthesis",
      "Arginine and proline metabolism","Aminoacyl-tRNA biosynthesis","D-Amino acid metabolism",
      "Glycine, serine and threonine metabolism","Histidine metabolism","Glutathione metabolism",
      "Lysine degradation","Lysine biosynthesis","Phenylalanine metabolism",
      "Phenylalanine, tyrosine and tryptophan biosynthesis","Purine metabolism",
      "Pyrimidine metabolism","Tryptophan metabolism","Tyrosine metabolism",
      "Cysteine and methionine metabolism","Valine, leucine and isoleucine biosynthesis",
      "Valine, leucine and isoleucine degradation","beta-Alanine metabolism",
      "Taurine and hypotaurine metabolism"
    ) ~ "Amino acid & nucleotide metabolism",

    x %in% c(
      "Biosynthesis of unsaturated fatty acids","Fatty acid biosynthesis","Fatty acid degradation",
      "Fatty acid elongation","Arachidonic acid metabolism","alpha-Linolenic acid metabolism",
      "Linoleic acid metabolism","Glycerolipid metabolism","Glycerophospholipid metabolism",
      "Sphingolipid metabolism","Ether lipid metabolism","Steroid biosynthesis",
      "Steroid hormone biosynthesis","Primary bile acid biosynthesis"
    ) ~ "Lipid metabolism",

    x %in% c(
      "Biotin metabolism","Folate biosynthesis","One carbon pool by folate",
      "Pantothenate and CoA biosynthesis","Porphyrin metabolism","Riboflavin metabolism",
      "Thiamine metabolism","Vitamin B6 metabolism","Nicotinate and nicotinamide metabolism",
      "Ubiquinone and other terpenoid-quinone biosynthesis"
    ) ~ "Cofactor & vitamin metabolism",

    TRUE ~ "Other metabolism"
  )
}

# ------------------------------------------------------------
# Load data and identify PIVC1 embryo samples
# ------------------------------------------------------------

dat <- read_excel(data_file) %>% as.data.frame(check.names=FALSE)
kegg_ref <- read.csv(kegg_file, stringsAsFactors=FALSE, check.names=FALSE)

if(!all(c("Compounds","cpd_ID") %in% colnames(dat))) stop("Input metabolite table requires Compounds and cpd_ID columns.")
if(!all(c("cpd_ID","Pathway") %in% colnames(kegg_ref))) stop("KEGG mapping table requires cpd_ID and Pathway columns.")
if(!"pathway_id" %in% colnames(kegg_ref)) kegg_ref$pathway_id <- ""

sample_pattern <- paste0("^(",paste(stage_order,collapse="|"),")[._-]?E[0-9]+$")
sample_cols <- colnames(dat)[str_detect(toupper(colnames(dat)),sample_pattern)]
if(length(sample_cols)==0) stop("No PIVC1 embryo sample columns were detected.")

stage_regex <- paste0("^(",paste(stage_order[order(nchar(stage_order),decreasing=TRUE)],collapse="|"),")")
sample_info <- data.frame(Sample=sample_cols,stringsAsFactors=FALSE) %>%
  mutate(Stage=str_extract(toupper(Sample),stage_regex),
         Stage=factor(Stage,levels=stage_order)) %>%
  filter(!is.na(Stage)) %>% arrange(Stage,Sample)

stage_counts <- table(factor(sample_info$Stage,levels=stage_order))
if(any(stage_counts==0)) stop("Missing stages: ",paste(names(stage_counts)[stage_counts==0],collapse=", "))

# ------------------------------------------------------------
# Abundance matrix -> log2(x+1) -> stage means -> row Z-scores
# ------------------------------------------------------------

mat_df <- dat %>%
  select(Compounds,all_of(sample_info$Sample)) %>%
  filter(!is.na(Compounds),Compounds!="") %>%
  mutate(across(all_of(sample_info$Sample),~suppressWarnings(as.numeric(.x)))) %>%
  group_by(Compounds) %>%
  summarise(across(all_of(sample_info$Sample),~mean(.x,na.rm=TRUE)),.groups="drop")

for(j in sample_info$Sample) mat_df[[j]][!is.finite(mat_df[[j]])] <- NA_real_

mat <- as.matrix(mat_df[,sample_info$Sample,drop=FALSE])
storage.mode(mat) <- "numeric"
rownames(mat) <- mat_df$Compounds
mat_log <- log2(mat+1)

stage_mean <- sapply(stage_order,function(st) {
  smp <- sample_info$Sample[as.character(sample_info$Stage)==st]
  rowMeans(mat_log[,smp,drop=FALSE],na.rm=TRUE)
})
rownames(stage_mean) <- rownames(mat_log)
colnames(stage_mean) <- stage_order
stage_mean[!is.finite(stage_mean)] <- NA_real_

row_sd <- apply(stage_mean,1,sd,na.rm=TRUE)
keep <- rowSums(is.finite(stage_mean))==length(stage_order) & is.finite(row_sd) & row_sd>0
stage_mean <- stage_mean[keep,,drop=FALSE]

stage_z <- t(scale(t(stage_mean)))
stage_z[!is.finite(stage_z)] <- NA_real_

# ------------------------------------------------------------
# Compound -> KEGG compound -> KEGG metabolic pathway
# ------------------------------------------------------------

cpd_raw <- data.frame(
  Compounds=as.character(dat$Compounds),
  cpd_raw=as.character(dat$cpd_ID),
  stringsAsFactors=FALSE
)
cpd_lists <- str_extract_all(toupper(cpd_raw$cpd_raw),"C[0-9]{5}")

map_met <- bind_rows(lapply(seq_len(nrow(cpd_raw)),function(i) {
  ids <- unique(cpd_lists[[i]])
  if(length(ids)==0) return(NULL)
  data.frame(Compounds=cpd_raw$Compounds[i],cpd_ID=ids,stringsAsFactors=FALSE)
})) %>%
  filter(!is.na(Compounds),Compounds!="",!is.na(cpd_ID),cpd_ID!="") %>%
  distinct() %>% filter(Compounds %in% rownames(stage_z))

path_cpd_lists <- str_extract_all(toupper(as.character(kegg_ref$cpd_ID)),"C[0-9]{5}")

pathway_map <- bind_rows(lapply(seq_len(nrow(kegg_ref)),function(i) {
  ids <- unique(path_cpd_lists[[i]])
  if(length(ids)==0) return(NULL)

  pw <- as.character(kegg_ref$Pathway[i])
  pw <- str_remove(pw,"\\s*-\\s*Mus musculus.*$")
  pw <- str_remove(pw,"\\s*-\\s*Homo sapiens.*$")
  pw <- str_squish(pw)

  data.frame(
    cpd_ID=ids,Pathway=pw,
    pathway_id=as.character(kegg_ref$pathway_id[i]),
    stringsAsFactors=FALSE
  )
})) %>%
  mutate(
    Pathway_code=str_extract(paste(pathway_id,Pathway),"[0-9]{5}"),
    Pathway_num=suppressWarnings(as.numeric(Pathway_code))
  ) %>%
  filter(!is.na(cpd_ID),cpd_ID!="",!is.na(Pathway),Pathway!="") %>%
  distinct()

if(any(is.finite(pathway_map$Pathway_num))) {
  pathway_map <- pathway_map %>%
    filter(is.finite(Pathway_num),Pathway_num>=10,Pathway_num<=1299)
}

pathway_map <- pathway_map %>%
  filter(
    !Pathway %in% c(
      "Metabolic pathways",
      "Biosynthesis of secondary metabolites",
      "Microbial metabolism in diverse environments"
    ),
    !str_detect(
      Pathway,
      regex(
        paste(c(
          "Drug metabolism","Metabolism of xenobiotics","Neomycin","Kanamycin",
          "Gentamicin","Streptomycin","Penicillin","Cephalosporin","Carbapenem","Monobactam"
        ),collapse="|"),
        ignore_case=TRUE
      )
    )
  )

compound_pathway <- map_met %>%
  inner_join(pathway_map %>% select(cpd_ID,Pathway) %>% distinct(),by="cpd_ID") %>%
  select(Compounds,Pathway) %>% distinct() %>%
  filter(Compounds %in% rownames(stage_z))

background_metabolites <- unique(compound_pathway$Compounds)
pathways_all <- unique(compound_pathway$Pathway)

# ------------------------------------------------------------
# Stage x pathway directional statistics
# ------------------------------------------------------------

res_list <- list()
ii <- 1L

for(pw in pathways_all) {
  pathway_mets <- unique(compound_pathway$Compounds[compound_pathway$Pathway==pw])
  pathway_mets <- intersect(pathway_mets,background_metabolites)
  if(length(pathway_mets)<min_pathway_metabolites) next

  background_other <- setdiff(background_metabolites,pathway_mets)

  for(st in stage_order) {
    x <- stage_z[pathway_mets,st]
    y <- stage_z[background_other,st]
    x <- x[is.finite(x)]
    y <- y[is.finite(y)]

    if(length(x)<min_pathway_metabolites || length(y)<10) next

    p_now <- tryCatch(
      wilcox.test(x,y,alternative="two.sided",exact=FALSE)$p.value,
      error=function(e) NA_real_
    )

    res_list[[ii]] <- data.frame(
      Stage=st,
      Pathway=pw,
      Metabolite_count=length(x),
      Mean_pathway_Z=mean(x),
      Mean_background_Z=mean(y),
      Directional_score=mean(x)-mean(y),
      P_value=p_now,
      stringsAsFactors=FALSE
    )
    ii <- ii+1L
  }
}

result_all <- bind_rows(res_list) %>%
  group_by(Stage) %>%
  mutate(FDR=p.adjust(P_value,method="BH")) %>%
  ungroup() %>%
  mutate(
    log10P=-log10(P_value+1e-300),
    Stage=factor(Stage,levels=stage_order)
  )

# ------------------------------------------------------------
# Pathway categories and figure selection
# ------------------------------------------------------------

result_global <- result_all %>%
  mutate(
    Category=factor(assign_category(Pathway),levels=category_order),
    Is_core=Pathway %in% core_pathways
  )

pathway_selection <- result_global %>%
  group_by(Pathway,Category) %>%
  summarise(
    Is_core=any(Is_core),
    Best_P=min(P_value,na.rm=TRUE),
    Max_abs_score=max(abs(Directional_score),na.rm=TRUE),
    .groups="drop"
  ) %>%
  filter(Is_core | Best_P<0.05)

plot_global <- result_global %>%

order_global <- pathway_selection %>%
  mutate(
    Core_order=match(Pathway,core_pathways),
    Core_order=ifelse(is.na(Core_order),999,Core_order)
  ) %>%
  arrange(Category,desc(Is_core),Core_order,desc(Max_abs_score),Best_P,Pathway)

plot_global <- plot_global %>%
  mutate(
    Stage=factor(Stage,levels=stage_order),
    Category=factor(Category,levels=category_order),
    Pathway=factor(Pathway,levels=rev(order_global$Pathway))
  )

colour_limit <- as.numeric(

if(!is.finite(colour_limit) || colour_limit<=0) {

if(!is.finite(colour_limit) || colour_limit<=0) colour_limit <- 1

if (RUN_G) {

if (RECOMPUTE_GSEA) {
  suppressPackageStartupMessages({
    library(clusterProfiler); library(org.Mm.eg.db); library(AnnotationDbi)
  })
run_gsea_kegg <- function(
    res_df,
    id_col = "Geneid",
    from_type = "ENSEMBL",
    score_col = "stat",
    organism = "mmu"
) {

  #----------------------------------------------------------
  # 1. 整理差异分析结果
  #----------------------------------------------------------

  df <- as.data.frame(
    res_df,
    stringsAsFactors = FALSE
  )

  if (!id_col %in% colnames(df)) {
    stop(
      "结果中没有找到基因ID列：",
      id_col
    )
  }

  # 若没有stat，则使用log2FoldChange
  if (!score_col %in% colnames(df)) {

    if ("log2FoldChange" %in% colnames(df)) {

      score_col <- "log2FoldChange"

      message(
        "没有找到stat，改用log2FoldChange排序。"
      )

    } else {

      stop(
        "结果中既没有stat，也没有log2FoldChange。"
      )
    }
  }

  df <- df %>%
    dplyr::filter(
      !is.na(.data[[id_col]]),
      .data[[id_col]] != "",
      !is.na(.data[[score_col]]),
      is.finite(.data[[score_col]])
    )

  # 去除ENSEMBL版本号
  df[[id_col]] <- sub(
    "\\..*$",
    "",
    as.character(df[[id_col]])
  )

  #----------------------------------------------------------
  # 2. ENSEMBL转换成ENTREZID
  #----------------------------------------------------------

  gene_map <- suppressMessages(
    clusterProfiler::bitr(
      unique(df[[id_col]]),
      fromType = from_type,
      toType = "ENTREZID",
      OrgDb = org.Mm.eg.db
    )
  )

  if (nrow(gene_map) == 0) {
    stop("没有基因成功转换为ENTREZID。")
  }

  # Geneid对应gene_map中的ENSEMBL
  join_by <- stats::setNames(
    from_type,
    id_col
  )

  df2 <- df %>%
    dplyr::inner_join(
      gene_map,
      by = join_by
    )

  #----------------------------------------------------------
  # 3. 一个ENTREZID只保留一个排序值
  #----------------------------------------------------------

  df2 <- df2 %>%
    dplyr::group_by(
      ENTREZID
    ) %>%
    dplyr::slice_max(
      order_by = abs(.data[[score_col]]),
      n = 1,
      with_ties = FALSE
    ) %>%
    dplyr::ungroup()

  #----------------------------------------------------------
  # 4. 构建GSEA排序向量
  #----------------------------------------------------------

  gene_list <- df2[[score_col]]

  names(gene_list) <- as.character(
    df2$ENTREZID
  )

  gene_list <- sort(
    gene_list,
    decreasing = TRUE
  )

  # 删除可能重复的名字
  gene_list <- gene_list[
    !duplicated(names(gene_list))
  ]

  message(
    "进入GSEA的基因数量：",
    length(gene_list)
  )

  #----------------------------------------------------------
  # 5. KEGG GSEA
  #----------------------------------------------------------

  gsea_kegg <- clusterProfiler::gseKEGG(
    geneList = gene_list,
    organism = organism,
    keyType = "ncbi-geneid",
    minGSSize = 10,
    maxGSSize = 500,
    pvalueCutoff = 1,
    pAdjustMethod = "BH",
    verbose = FALSE,
    seed = TRUE
  )

  gsea_df <- as.data.frame(
    gsea_kegg
  )

  list(
    gsea = gsea_kegg,
    table = gsea_df,
    gene_list = gene_list,
    mapped_table = df2,
    score_col = score_col
  )
}
  obj <- readRDS("data/DESeq2_results_Embryo_YolkSac.rds")
  # DESeq2 results must be tissue-specific 2DG-versus-Control results.
  # Original function uses stat, falling back to log2FoldChange if absent.
  gsea_g <- bind_rows(lapply(c("Embryo", "Yolk_sac"), function(tissue) {
    res <- obj[[paste0("res_", tissue)]][["2DG_vs_Control"]]
    if (is.null(res)) stop("Missing tissue-specific 2DG_vs_Control result")
    ans <- run_gsea_kegg(res)$table
    ans$Treatment <- "2DG"
    ans$Tissue <- tissue
    ans
  }))
  write.csv(gsea_g, "results/ExtendedData_Figure2g_GSEA_all.csv", row.names=FALSE)
} else {
  gsea_g <- read.csv("data/KEGG_GSEA_all_results.csv", check.names=FALSE)
}
required <- c("Treatment", "Tissue", "Description", "NES", "p.adjust")
if (!all(required %in% names(gsea_g))) stop("GSEA table requires: ", paste(required, collapse=", "))
if (anyDuplicated(gsea_g[, c("Treatment", "Tissue", "Description")])) {
  stop("Duplicate pathway rows within a treatment/tissue; resolve input explicitly")
}
plot_g <- gsea_g %>%
  filter(Treatment == "2DG", Tissue %in% c("Embryo", "Yolk sac", "Yolk_sac"),
         is.finite(NES), NES > 0, is.finite(p.adjust)) %>%
  mutate(Tissue=factor(ifelse(Tissue == "Embryo", "Embryo", "YS"),
                       levels=c("Embryo", "YS")),
         Description_clean=str_remove(Description, " - Mus musculus.*$"),
         neg_log10_FDR=-log10(pmax(p.adjust, .Machine$double.xmin))) %>%
  group_by(Tissue) %>% slice_max(NES, n=10, with_ties=FALSE) %>% ungroup() %>%
  arrange(Tissue, NES) %>%
  mutate(Pathway_group=paste(Description_clean, Tissue, sep="___"))
if (nrow(plot_g)==0 || length(unique(plot_g$Tissue))!=2) stop("Both tissues need positive 2DG GSEA results")
plot_g$Pathway_group <- factor(plot_g$Pathway_group, levels=unique(plot_g$Pathway_group))
p_g <- ggplot(plot_g, aes(NES, Pathway_group, fill=neg_log10_FDR)) +
  geom_col(width=0.72, color="black", linewidth=0.25) +
  facet_wrap(~Tissue, scales="free_y", ncol=2,
             labeller=labeller(Tissue=c(Embryo="2-DG Embryo", YS="2-DG YS"))) +
  scale_y_discrete(labels=function(x) sub("___.*$", "", x)) +
  scale_fill_gradient(low="#D9D9D9", high="#FF2D2D", name=expression(-log[10](FDR))) +

  theme_bw(base_size=10) +
  theme(panel.grid.major.y=element_blank(), panel.grid.minor=element_blank(),
        axis.text=element_text(color="black"), strip.background=element_blank(),
        panel.spacing.x=grid::unit(1.5,"cm"))
print(p_g)
ggsave("results/ExtendedData_Figure2g.pdf", p_g, width=12, height=4.6)
write.csv(plot_g, "results/ExtendedData_Figure2g_displayed_pathways.csv", row.names=FALSE)
}
